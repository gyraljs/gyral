import { define, html, each, type TemplateResult } from '@gyral/core';

type Item = { id: number; title: string; done: boolean };
type Filter = 'All' | 'Active' | 'Completed';
type State = { items: Item[]; draft: string; filter: Filter; nextId: number };
type Msg = { _tag: 'Draft'; value: string } | { _tag: 'Add' } |
  { _tag: 'Toggle'; id: number; index: number } | { _tag: 'Delete'; id: number } |
  { _tag: 'Filter'; filter: Filter };
const filters: Filter[] = ['All', 'Active', 'Completed'];
const visible = (s: State) => s.items.filter(item => s.filter === 'All' || item.done === (s.filter === 'Completed'));

const App = define<State, Msg>()('pilot-app', {
  init: () => ({ items: [], draft: '', filter: 'All', nextId: 1 }),
  intent: {
    Draft: ({ value }) => ({ _tag: 'Draft', value: value ?? '' }),
    Add: true,
    Toggle: ({ target }) => ({ _tag: 'Toggle', id: Number(target.getAttribute('data-id')), index: Number(target.getAttribute('data-index')) }),
    Delete: ({ target }) => ({ _tag: 'Delete', id: Number(target.getAttribute('data-id')) }),
    Filter: ({ target }) => ({ _tag: 'Filter', filter: target.textContent as Filter }),
  },
  update: {
    Draft: (s, msg) => ({ ...s, draft: msg.value }),
    Add: s => s.draft.trim() ? { ...s, draft: '', nextId: s.nextId + 1, items: [...s.items, { id: s.nextId, title: s.draft.trim(), done: false }] } : s,
    Toggle: (s, msg) => {
      const id = __TOGGLE_ID__;
      return { ...s, items: s.items.map(item => item.id === id ? { ...item, done: !item.done } : item) };
    },
    Delete: (s, msg) => ({ ...s, items: s.items.filter(item => item.id !== msg.id) }),
    Filter: (s, msg) => ({ ...s, filter: msg.filter }),
  },
  view: (s, i) => html`
    <h1>Task list</h1>
    <form data-intent=${i.Add}>
      <label for="new-task">New task</label>
      <input id="new-task" value=${s.draft} data-intent=${i.Draft}>
      <button type="submit">Add task</button>
    </form>
    <nav aria-label="Task filters">
      ${filters.map(filter => html`<button type="button" data-intent=${i.Filter} aria-pressed=${s.filter === filter}>${filter}</button>`)}
    </nav>
    <p role="status">${s.items.filter(item => !item.done).length} active tasks</p>
    <ul>
      ${each(visible(s), item => item.id, (item, picked: { index: number; toggle: string; remove: string }): TemplateResult => html`
        <li>
          <label><input type="checkbox" ?checked=${item.done} data-id=${item.id} data-index=${picked.index} data-intent=${picked.toggle}>${item.title}</label>
          <button type="button" data-id=${item.id} data-intent=${picked.remove} aria-label=${`Delete ${item.title}`}>Delete</button>
        </li>
      `, item => ({ index: visible(s).indexOf(item), toggle: i.Toggle, remove: i.Delete }))}
    </ul>`
});
document.querySelector('main')?.append(new App());

import { define, each, html, intentsOf, type TemplateResult } from '@gyral/core';

export interface Item {
  readonly id: number;
  readonly title: string;
  readonly done: boolean;
}

export type Filter = 'All' | 'Active' | 'Completed';

export interface State {
  readonly items: readonly Item[];
  readonly draft: string;
  readonly filter: Filter;
  readonly nextId: number;
}

export type Msg =
  | { readonly _tag: 'Draft'; readonly value: string }
  | { readonly _tag: 'Add' }
  | { readonly _tag: 'Toggle'; readonly id: number }
  | { readonly _tag: 'Delete'; readonly id: number }
  | { readonly _tag: 'Filter'; readonly filter: Filter };

const filters: readonly Filter[] = ['All', 'Active', 'Completed'];

const visible = (s: State): readonly Item[] =>
  s.items.filter((item) => s.filter === 'All' || item.done === (s.filter === 'Completed'));

// The intent names as a module constant, so rows can use them and stay pure.
const i = intentsOf<typeof App>();

// A row: reads only its item, what `pick` returned (the id its checkbox toggles), and
// module constants.
const Row = (item: Item, toggleId: number | undefined): TemplateResult =>
  html`<li>
    <label>
      <input type="checkbox" value=${toggleId} ?checked=${item.done} data-intent=${i.Toggle} />
      ${item.title}
    </label>
    <button type="button" value=${item.id} data-intent=${i.Delete} aria-label=${`Delete ${item.title}`}>
      Delete
    </button>
  </li>`;

const idOf = (value: string | undefined): number | undefined => {
  const id = Number(value);
  return Number.isInteger(id) ? id : undefined;
};

const isFilter = (value: string | undefined): value is Filter =>
  filters.some((filter) => filter === value);

export const App = define<State, Msg>()('pilot-app', {
  init: () => ({ items: [], draft: '', filter: 'All', nextId: 1 }),
  intent: {
    Draft: ({ value }) => ({ _tag: 'Draft', value: value ?? '' }),
    Add: true,
    Toggle: ({ value }) => {
      const id = idOf(value);
      return id === undefined ? undefined : { _tag: 'Toggle', id };
    },
    Delete: ({ value }) => {
      const id = idOf(value);
      return id === undefined ? undefined : { _tag: 'Delete', id };
    },
    Filter: ({ value }) => (isFilter(value) ? { _tag: 'Filter', filter: value } : undefined),
  },
  update: {
    Draft: (s, m) => ({ ...s, draft: m.value }),
    Add: (s) => {
      const title = s.draft.trim();
      return title === ''
        ? s
        : {
            ...s,
            draft: '',
            nextId: s.nextId + 1,
            items: [...s.items, { id: s.nextId, title, done: false }],
          };
    },
    Toggle: (s, m) => ({
      ...s,
      items: s.items.map((item) => (item.id === m.id ? { ...item, done: !item.done } : item)),
    }),
    Delete: (s, m) => ({ ...s, items: s.items.filter((item) => item.id !== m.id) }),
    Filter: (s, m) => ({ ...s, filter: m.filter }),
  },
  view: (s, i) => {
    const shown = visible(s);
    return html`
      <h1>Task list</h1>
      <form data-intent=${i.Add}>
        <label for="new-task">New task</label>
        <input id="new-task" value=${s.draft} data-intent=${i.Draft} />
        <button type="submit">Add task</button>
      </form>
      <nav aria-label="Task filters">
        ${filters.map(
          (filter) =>
            html`<button type="button" value=${filter} data-intent=${i.Filter} aria-pressed=${s.filter === filter}>
              ${filter}
            </button>`,
        )}
      </nav>
      <p role="status">${s.items.filter((item) => !item.done).length} active tasks</p>
      <ul>
        ${each(shown, (item) => item.id, Row, (item) => __TOGGLE_ID__)}
      </ul>
    `;
  },
});

document.querySelector('main')?.append(new App());
